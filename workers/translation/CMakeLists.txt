cmake_minimum_required(VERSION 3.20)
project(blabber_translation LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
if(NOT LLAMA_SOURCE)
  message(FATAL_ERROR "LLAMA_SOURCE must point to the pinned llama.cpp checkout")
endif()
add_subdirectory(${LLAMA_SOURCE} llama)
add_executable(blabber-translation-worker main.cpp)
target_include_directories(blabber-translation-worker PRIVATE ${LLAMA_SOURCE}/vendor/nlohmann)
target_link_libraries(blabber-translation-worker PRIVATE llama)
install(TARGETS blabber-translation-worker RUNTIME DESTINATION .)
