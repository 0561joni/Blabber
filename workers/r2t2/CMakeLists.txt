cmake_minimum_required(VERSION 3.20)
project(blabber_r2t2 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT AUDIOCPP_SOURCE)
  message(FATAL_ERROR "AUDIOCPP_SOURCE must point to the pinned patched source")
endif()
set(AUDIOCPP_MODEL_SET custom CACHE STRING "" FORCE)
set(AUDIOCPP_MODELS confucius4_r2t2 CACHE STRING "" FORCE)
set(AUDIOCPP_DEPLOYMENT_BUILD ON CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_NATIVE_CPU OFF CACHE BOOL "" FORCE)
set(ENGINE_ENABLE_OPENMP OFF CACHE BOOL "" FORCE)
set(ENGINE_BUILD_MODEL_TESTS ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
add_subdirectory(${AUDIOCPP_SOURCE} audio EXCLUDE_FROM_ALL)
add_executable(blabber-r2t2-worker main.cpp ${AUDIOCPP_SOURCE}/src/capi/audiocpp.cpp)
target_include_directories(blabber-r2t2-worker PRIVATE ${AUDIOCPP_SOURCE}/include)
target_link_libraries(blabber-r2t2-worker PRIVATE engine_runtime cjson_vendor)
add_executable(blabber-r2t2-rolling-test rolling_test.cpp)
target_compile_options(blabber-r2t2-rolling-test PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME rolling-transcript COMMAND blabber-r2t2-rolling-test)
